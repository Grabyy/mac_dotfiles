void
dwiic_acpi_rescan(struct device *self)
{
	struct dwiic_softc *sc = (struct dwiic_softc *)self;

	aml_find_node(sc->sc_devnode, "_HID", dwiic_acpi_found_hid, sc);
}

else if (dwiic_matchhids(dev, ietp_hids) ||
    dwiic_matchhids(cdev, ietp_hids)) {
	if (crs.gpio_int_node != NULL &&
	    crs.gpio_int_node->gpio == NULL) {
		printf("dwiic: deferring IETP\n");
		config_defer(&sc->sc_dev, dwiic_acpi_rescan);
		return 0;
	}

	return dwiic_acpi_found_ietp(sc, node, dev, crs);
}
